```python
# Copyright (c) 2026, DCNET Cloud and contributors
# For license information, please see license.txt

import frappe
from frappe import _


def execute(filters=None):
    filters = frappe._dict(filters or {})
    columns = get_columns()
    data = get_data(filters)
    chart = get_chart(data)
    return columns, data, None, chart


def get_columns():
    return [
        {
            "fieldname": "source",
            "label": _("Nguon KH"),
            "fieldtype": "Data",
            "width": 200,
        },
        {
            "fieldname": "revenue",
            "label": _("Doanh so"),
            "fieldtype": "Currency",
            "width": 150,
        },
    ]


def get_data(filters):
    conditions = ["si.docstatus = 1"]
    values = {}

    if filters.get("from_date"):
        conditions.append("si.posting_date >= %(from_date)s")
        values["from_date"] = filters.from_date

    if filters.get("to_date"):
        conditions.append("si.posting_date <= %(to_date)s")
        values["to_date"] = filters.to_date

    if filters.get("company"):
        conditions.append("si.company = %(company)s")
        values["company"] = filters.company

    return frappe.db.sql(
        """
        SELECT
            IFNULL(c.source, 'Khong ro') as source,
            SUM(si.grand_total) as revenue
        FROM `tabSales Invoice` si
        LEFT JOIN `tabCustomer` c ON si.customer = c.name
        WHERE {conditions}
        GROUP BY c.source
        ORDER BY revenue DESC
        """.format(conditions=" AND ".join(conditions)),
        values,
        as_dict=1,
    )


def get_chart(data):
    if not data:
        return None
    return {
        "data": {
            "labels": [d.source for d in data],
            "datasets": [{"name": _("Doanh so"), "values": [d.revenue for d in data]}],
        },
        "type": "bar",
    }
```

```python
# Copyright (c) 2026, DCNET Cloud and contributors
# For license information, please see license.txt

import frappe
from frappe import _


def execute(filters=None):
    filters = frappe._dict(filters or {})
    columns = get_columns()
    data = get_data(filters)
    chart = get_chart(data)
    return columns, data, None, chart


def get_columns():
    return [
        {
            "fieldname": "item_code",
            "label": _("Ma SP"),
            "fieldtype": "Link",
            "options": "Item",
            "width": 150,
        },
        {
            "fieldname": "item_name",
            "label": _("Ten SP"),
            "fieldtype": "Data",
            "width": 250,
        },
        {
            "fieldname": "qty",
            "label": _("SL ban"),
            "fieldtype": "Float",
            "width": 100,
        },
        {
            "fieldname": "revenue",
            "label": _("Doanh so"),
            "fieldtype": "Currency",
            "width": 150,
        },
    ]


def get_data(filters):
    limit = filters.get("limit", 20)
    conditions = ["si.docstatus = 1"]
    values = {"limit": limit}

    if filters.get("from_date"):
        conditions.append("si.posting_date >= %(from_date)s")
        values["from_date"] = filters.from_date

    if filters.get("to_date"):
        conditions.append("si.posting_date <= %(to_date)s")
        values["to_date"] = filters.to_date

    if filters.get("company"):
        conditions.append("si.company = %(company)s")
        values["company"] = filters.company

    return frappe.db.sql(
        """
        SELECT
            sii.item_code,
            sii.item_name,
            SUM(sii.qty) as qty,
            SUM(sii.amount) as revenue
        FROM `tabSales Invoice Item` sii
        JOIN `tabSales Invoice` si ON sii.parent = si.name
        WHERE {conditions}
        GROUP BY sii.item_code, sii.item_name
        ORDER BY revenue DESC
        LIMIT %(limit)s
        """.format(conditions=" AND ".join(conditions)),
        values,
        as_dict=1,
    )


def get_chart(data):
    if not data:
        return None
    return {
        "data": {
            "labels": [d.item_name for d in data[:10]],
            "datasets": [
                {"name": _("Doanh so"), "values": [d.revenue for d in data[:10]]}
            ],
        },
        "type": "bar",
    }
```
